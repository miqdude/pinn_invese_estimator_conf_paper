\section{System Description}

\subsection{System Modeling}

\begin{figure}[htbp]
    \centering
    \includegraphics[width=0.8\linewidth]{images/frame_setup.png}
    \caption{Illustration of frame setup from top view of $W$. The rigid frame is a board with size of $1\times 1$ meter. Four quadrotor attached at position of $\pm [0.5, 0.5]$ from the center of $\mathbf{p}$.}
    \label{fig:frame_setup}
\end{figure}

We define an inertial world frame $W$ with an upward-pointing $Z$-axis. The rigid frame vehicle consists of a board with position $\mathbf{p} = [\mathbf{p}_x, \mathbf{p}_y, \mathbf{p}_z]^T$ and orientation (roll, pitch, yaw) $\boldsymbol{\eta} = [\phi, \theta, \psi]^T$ on $W$ with mass $m_p$. At the corners of $\mathbf{p}$ attached Iris Quadrotor with position $\mathbf{q}$ and index $i$ on $\mathbf{p}$ with mass $m_q$ \cite{Furrer2016}. The payload's position on $\mathbf{p}$ is defined by $\mathbf{p}_p = [\mathbf{p}_x, \mathbf{p}_y, \mathbf{p}_z]$ with mass $m_{pp}$.

The MAS controller employs a centralized control architecture derived from \cite{mellinger2013trajectory}. The composite rigid body is commanded via a unified control wrench $\boldsymbol{\tau}$, defined as:

\begin{equation}
    \boldsymbol{\tau} = [F, M_{x}, M_{y}, M_{z}]^T,
\end{equation}

\noindent where $F$ is the total collective thrust along the board's $Z$-axis, and $M_{x}$, $M_{y}$, and $M_{z}$ are the desired roll, pitch, and yaw moments applied to the board frame $\mathbf{p}$. These relate to the individual quadrotor thrust $F_i$ with thrust input $\mathbf{u} = [F_1, F_2, F_3, F_4]^T$, where $F_i$ denotes the thrust produced by quadrotor along its local $Z$-axis, through the allocation matrix $A \in \mathbb{R}^{4 \times 4}$:

\begin{equation} \label{eq:tau}
    \boldsymbol{\tau} = A \mathbf{u},
\end{equation}
\begin{equation}
    A =
    \begin{bmatrix}
        1 & 1 & 1 & 1 \\
        y_1 & y_2 & y_3 & y_4 \\
        -x_1 & -x_2 & -x_3 & -x_4 \\
        \gamma_1 & \gamma_2 & \gamma_3 & \gamma_4
    \end{bmatrix}.
\end{equation}

The rows of $A$ correspond to the total thrust, roll moment, pitch moment, and yaw moment on the board, respectively. Here, $(x_i, y_i)$ is the position of quadrotor $i$'s center of thrust expressed in $\mathcal{B}$, and $\gamma_i$ is the yaw moment coefficient (torque-to-thrust ratio) of quadrotor $i$, with its sign dictated by the rotor's spin direction.

\subsection{Dynamic Thrust Allocation}

When a payload shifts from the board's geometric center, it alters the
effective moment arms of each quadrotor. Let $\hat{\mathbf{c}} =
[\hat{c}_x, \hat{c}_y]^T$ denote the estimated system CoG expressed in
$\mathbf{p}$, produced directly by the PINN and fed to the control
allocation matrix. Redefining the lever arms of quadrotor $i$, measured from
the system CoG, as $\Delta x_i = x_i - \hat{c}_x$ and $\Delta y_i = y_i -
\hat{c}_y$, the allocation matrix adapts dynamically:

\begin{equation}
    \mathbf{A}_{\text{final}} =
    \begin{bmatrix}
        1 & 1 & 1 & 1 \\
        \Delta y_1 & \Delta y_2 & \Delta y_3 & \Delta y_4 \\
        -\Delta x_1 & -\Delta x_2 & -\Delta x_3 & -\Delta x_4 \\
        \gamma_1 & \gamma_2 & \gamma_3 & \gamma_4
    \end{bmatrix}.
    \label{eq:allocation_matrix}
\end{equation}

\noindent Control inputs are computed via the Moore-Penrose pseudo-inverse to
handle transient rank deficiencies:

\begin{equation}
    \mathbf{u} = \mathbf{A}_{\text{final}}^{+}\boldsymbol{\tau}.
    \label{eq:final_u}
\end{equation}

\subsection{Centralized Control Design}
Based on \cite{mellinger2013trajectory}, the MAS controller separates position and attitude tracking. Position and velocity errors define commanded acceleration $\mathbf{\ddot{r}}_{cmd}$:

\begin{equation}
    \mathbf{e}_p = \mathbf{p}_{des} - \mathbf{p}, \quad
    \mathbf{e}_v = \mathbf{\dot{p}}_{des} - \mathbf{\dot{p}},
\end{equation}

\begin{equation}\mathbf{\ddot{p}}_{cmd} = K_p \mathbf{e}_p + K_d \mathbf{e}_v + \mathbf{\ddot{p}}_{\text{ff}}.
\end{equation}

Desired roll ($\phi^{des}$), pitch ($\theta^{des}$), and total thrust ($F^{des}$) are extracted via small-angle approximation:

\begin{equation}
    \begin{split}
        \phi^{des} &= \frac{1}{g} (\ddot{p}^{des}_{x} \sin \psi_T - \ddot{p}^{des}_{y} \cos \psi_T), \\
        \theta^{des} &= \frac{1}{g} (\ddot{r}^{des}_{y} \cos \psi_T + \ddot{p}^{des}_{y} \sin \psi_T), \\
        F^{des} &= m(\ddot{p}^{des}_{z} + g).
    \end{split}
\end{equation}

Inner-loop desired moments $\mathbf{M}^{des}$ regulate orientation against angular velocities $\omega = (p, q, r)$:

\begin{equation}
    \begin{split}        
        M_{x}^{des} &= k_{p, \phi} (\phi^{des} - \phi) + k_{d, \phi} (p^{des} - p), \\
        M_{y}^{des} &= k_{p, \theta} (\theta^{des} - \theta) + k_{d, \theta} (q^{des} - q), \\
        M_{z}^{des} &= k_{p, \psi} (\psi^{des} - \psi) + k_{d, \psi} (r^{des} - r).
    \end{split}
\end{equation}
